  const [msg, setMsg] = React.useState(0);
  const [pct, setPct] = React.useState(0);
  const MSGS = ['Scanning the pitch…','Squads located','Possession mapped'];
  React.useEffect(() => {
    const t = setTimeout(()=>setGone(true), 3400);
    const m = setInterval(()=>setMsg(i=>Math.min(i+1, MSGS.length-1)), 1000);
    const p = setInterval(()=>setPct(v=>Math.min(v+Math.ceil(Math.random()*4)+1, 100)), 90);
    return () => { clearTimeout(t); clearInterval(m); clearInterval(p); };
  }, []);
  if (gone) return null;

  const blip = (x,y,delay) => (
    <span style={{position:'absolute',left:x+'%',top:y+'%',width:7,height:7,marginLeft:-3.5,marginTop:-3.5,borderRadius:'50%',background:'#C9F9D9',boxShadow:'0 0 10px 3px rgba(74,222,128,.8)',animation:'ccBlip 2s linear '+delay+'s infinite'}}></span>
  );
  return (
    <div style={{position:'absolute',inset:0,zIndex:600,background:'radial-gradient(120% 90% at 50% 38%, #101B14 0%, #0A120D 70%)',display:'flex',flexDirection:'column',alignItems:'center',justifyContent:'center',gap:24,overflow:'hidden',animation:'ccSplashOut .55s ease 2.85s forwards'}}>

      {[{l:10,t:10,bl:0,bt:1,br:1},{r:10,t:10},{l:10,b:10},{r:10,b:10}].map((c,i)=>(
        <span key={i} style={{position:'absolute',width:22,height:22,left:c.l,right:c.r,top:c.t,bottom:c.b,borderColor:'rgba(74,222,128,.5)',borderStyle:'solid',borderWidth:(c.t!=null?'1.5px':0)+' '+(c.r!=null?'1.5px':0)+' '+(c.b!=null?'1.5px':0)+' '+(c.l!=null?'1.5px':0)}}></span>
      ))}
      <div style={{position:'absolute',top:14,left:42,fontFamily:'ui-monospace,SFMono-Regular,Menlo,monospace',fontSize:8.5,letterSpacing:'1px',color:'rgba(74,222,128,.75)',lineHeight:1.6}}>CC-OS v2.6<br/>BOOT SEQUENCE</div>

function App() {
  const [dev, setDev] = React.useState('phone');
  const [loc, setLoc] = React.useState('home');
  const [board, setBoard] = React.useState(false);
  const [locked, setLocked] = React.useState(false);
  const [capture, setCapture] = React.useState(false);
  const [apps, setApps] = React.useState(false);
  const [emg, setEmg] = React.useState(false);
  const phase = 'mid';
  const overlayOpen = board || locked || capture || apps;
  const upVars = { '--home-text2':'#6E5636', '--office-text2':'#46577A', '--outside-text2':'#3F6E52' };
  const wide = dev==='ipad';

  return (
    <div style={{display:'flex',flexDirection:'column',alignItems:'center',gap:14,width:'100%'}}>
    <div style={{display:'flex',gap:6,background:'rgba(0,0,0,.07)',borderRadius:12,padding:4}}>
      {[['phone','📱 Phone'],['ipad','💻 iPad']].map(([v,l]) => (
        <button key={v} onClick={()=>setDev(v)} style={{border:'none',borderRadius:9,padding:'6px 16px',fontSize:12,fontWeight:600,cursor:'pointer',fontFamily:'var(--font-body)',background:dev===v?'#fff':'transparent',color:dev===v?'#2D1B0E':'#6E5636',boxShadow:dev===v?'0 2px 8px rgba(0,0,0,.12)':'none'}}>{l}</button>
      ))}
    </div>
    <div
      data-theme={loc==='home'?undefined:loc}
      data-screen-label={loc==='office'?'Office Dashboard':'Pitch Home (dockless)'}
      style={{
        width:'100%',maxWidth:wide?900:'var(--app-max-width)',minHeight:720,background:'var(--bg)',color:'var(--text)',
        borderRadius:28,boxShadow:'0 24px 64px rgba(0,0,0,.25)',position:'relative',overflow:'hidden',
        transition:'background .5s,color .4s', ...upVars,
      }}>
      {loc==='office'
        ? <OfficeDashboard onQuickAdd={()=>setCapture(true)} />
        : <PitchHome loc={loc} setLoc={setLoc} phase={phase} overlayOpen={overlayOpen} onQuickAdd={()=>setCapture(true)} onOpenCoach={()=>setBoard(true)} onOpenApps={()=>setApps(true)} />}
      {loc==='office' && (
        <div style={{padding:'0 16px 16px'}}>
          <SegmentedControl
            options={[{value:'home',label:'🏠 Home'},{value:'office',label:'🏢 Office'},{value:'outside',label:'🌳 Outside'}]}
            value={loc} onChange={setLoc} />
        </div>
      )}
      {/* Lone emergency button — the only floating control */}
      <button onClick={()=>setEmg(e=>!e)} aria-label="Emergency mode" title="Emergency mode" style={{position:'absolute',bottom:16,right:16,zIndex:400,width:48,height:48,borderRadius:'50%',border:'1.5px solid '+(emg?'#DC2626':'var(--as)'),cursor:'pointer',display:'flex',alignItems:'center',justifyContent:'center',fontSize:17,background:emg?'#DC2626':'var(--card)',boxShadow:emg?'0 0 0 4px rgba(220,38,38,.22), 0 8px 24px rgba(220,38,38,.3)':'0 8px 24px rgba(0,0,0,.16)',transition:'all .2s'}}>🚨</button>
      <Splash />
      <QuickCapture open={capture} onClose={()=>setCapture(false)} />
      <SquadBoard key={dev} open={board} onClose={()=>setBoard(false)} phase={phase} wide={wide} />
      <MyApps open={apps} onClose={()=>setApps(false)} />
      <LockScreen open={locked} onUnlock={()=>setLocked(false)} />
    </div>
    </div>
  );
}
ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
