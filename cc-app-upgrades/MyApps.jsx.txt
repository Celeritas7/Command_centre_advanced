// My Apps — portal grouped by squad (Attackers / Midfielders / Defenders), with categories inside each squad
const { TextInput, Chip } = window.CommandCentreDesignSystem_63bc0b;

const MA_SEED = [
  { key:'atk', icon:'⚔️', name:'Attackers', color:'#DC2626', sub:'deep work & growth', cats:[
    { id:'lang', name:'Language Study', color:'#38A169', apps:[
      { id:'b1', name:'JP Grammar', icon:'📈', on:true },
      { id:'b2', name:'JLPT study app', icon:'🈴', on:true },
      { id:'c1', name:'Chinese study app', icon:'🀄', on:true },
      { id:'a8', name:'Scriptura_app', icon:'📜', on:true },
    ]},
    { id:'fin', name:'Finance', color:'#B7791F', apps:[
      { id:'a2', name:'Cost Management', icon:'💰', on:true },
      { id:'a9', name:'Investment', icon:'📊', on:true },
    ]},
    { id:'car', name:'Career', color:'#3B5BDB', apps:[
      { id:'a5', name:'Roadmap', icon:'🗺️', on:true },
      { id:'a7', name:'writing_test', icon:'✍️', on:true },
    ]},
  ]},
  { key:'mid', icon:'🏃', name:'Midfielders', color:'#D97706', sub:'routines & daily ops', cats:[
    { id:'ops', name:'Daily Ops', color:'#D97706', apps:[
      { id:'a1', name:'Calendar', icon:'📅', on:false },
      { id:'a3', name:'Food department', icon:'🍱', on:true },
      { id:'a6', name:'Weekly_focus', icon:'🎯', on:true },
    ]},
  ]},
  { key:'def', icon:'🪞', name:'Defenders', color:'#7C3AED', sub:'reflection & wind-down', cats:[
    { id:'ref', name:'Reflection', color:'#7C3AED', apps:[
      { id:'a4', name:'Commonplace', icon:'📓', on:true },
    ]},
  ]},
];

function MATile({ app, edit, groupColor, onToggle, onRename, onRemove, onLaunch }) {
  return (
    <div style={{position:'relative',display:'flex',flexDirection:'column',alignItems:'center',gap:5,minWidth:0}}>
      <button onClick={edit?onToggle:onLaunch} title={edit?(app.on?'Tap to deactivate':'Tap to activate'):('Open '+app.name)} style={{
        width:40,height:40,borderRadius:12,border:'1.5px solid '+(app.on?groupColor+'33':'var(--as)'),cursor:'pointer',
        background:app.on?'var(--card)':'var(--as)',fontSize:18,display:'flex',alignItems:'center',justifyContent:'center',
        boxShadow:app.on?'0 2px 8px rgba(0,0,0,.07)':'none',opacity:app.on?1:.45,filter:app.on?'none':'grayscale(.8)',
        transition:'all .2s',position:'relative',
      }}>
        {app.icon}
        <span style={{position:'absolute',bottom:2,right:2,width:7,height:7,borderRadius:'50%',background:app.on?'#16A34A':'#A8A29E',border:'1.5px solid #fff'}}></span>
      </button>
      {edit && (
        <button onClick={onRemove} aria-label="Remove" style={{position:'absolute',top:-6,left:-4,width:18,height:18,borderRadius:'50%',border:'none',background:'#57534E',color:'#fff',fontSize:11,cursor:'pointer',lineHeight:1,zIndex:2}}>×</button>
      )}
      <button onClick={edit?onRename:onLaunch} style={{border:'none',background:'transparent',padding:0,cursor:'pointer',fontFamily:'var(--font-body)',fontSize:9,fontWeight:600,color:app.on?'var(--text)':'var(--text2)',maxWidth:'100%',overflow:'hidden',textOverflow:'ellipsis',whiteSpace:'nowrap',textDecoration:edit?'underline dotted':'none'}}>{app.name}</button>
    </div>
  );
}

function MyApps({ open, onClose }) {
  const [squads, setSquads] = React.useState(MA_SEED);
  const [edit, setEdit] = React.useState(false);
  const [q, setQ] = React.useState('');
  const [filter, setFilter] = React.useState('all');
  if (!open) return null;
  const allApps = squads.flatMap(s=>s.cats.flatMap(c=>c.apps));
  const total = allApps.length, active = allApps.filter(a=>a.on).length;
  function mutateCat(sk, cid, fn) {
    setSquads(ss => ss.map(s => s.key===sk ? { ...s, cats:s.cats.map(c => c.id===cid ? fn(c) : c) } : s));
  }
  function addApp(sk, cid) {
    const name = window.prompt('App name?');
    if (!name || !name.trim()) return;
    mutateCat(sk, cid, c => ({ ...c, apps:[...c.apps, { id:'n'+Date.now(), name:name.trim(), icon:'🔗', on:true }] }));
  }
  function addCat(sk) {
    const name = window.prompt('Category name?');
    if (!name || !name.trim()) return;
    setSquads(ss => ss.map(s => s.key===sk ? { ...s, cats:[...s.cats, { id:'c'+Date.now(), name:name.trim(), color:s.color, apps:[] }] } : s));
  }
  const ql = q.trim().toLowerCase();
  const visible = a => (filter==='all' || (filter==='on')===a.on) && (!ql || a.name.toLowerCase().includes(ql));
  return (
    <div style={{position:'absolute',inset:0,zIndex:300,background:'var(--bg)',display:'flex',flexDirection:'column',overflow:'auto'}} data-screen-label="My Apps">
      <div style={{position:'sticky',top:0,zIndex:10,background:'var(--bg)',padding:'16px 16px 10px',display:'flex',flexDirection:'column',gap:10}}>
        <div style={{display:'flex',alignItems:'center',gap:8}}>
          <div style={{fontFamily:'var(--font-display)',fontSize:17,fontWeight:700}}>My Apps</div>
          <span style={{fontSize:10,fontWeight:700,color:'var(--accent)',background:'var(--as)',borderRadius:8,padding:'2px 8px'}}>{active} / {total} active</span>
          <button onClick={()=>setEdit(e=>!e)} style={{marginLeft:'auto',border:'1.5px solid '+(edit?'var(--accent)':'var(--as)'),background:edit?'var(--accent)':'var(--card)',color:edit?'#fff':'var(--text)',borderRadius:9,padding:'5px 14px',fontSize:11,fontWeight:700,cursor:'pointer',fontFamily:'var(--font-body)',whiteSpace:'nowrap'}}>{edit?'✓ Done':'✎ Edit'}</button>
          <button onClick={onClose} aria-label="Close" style={{border:'none',background:'var(--as)',width:30,height:30,borderRadius:9,fontSize:14,cursor:'pointer',color:'var(--text)',flexShrink:0}}>×</button>
        </div>
        <div style={{display:'flex',gap:6,alignItems:'center'}}>
          <TextInput placeholder="Search apps…" value={q} onChange={(e)=>setQ(e.target.value)} style={{flex:1}} />
          <Chip active={filter==='all'} onClick={()=>setFilter('all')}>All</Chip>
          <Chip active={filter==='on'} onClick={()=>setFilter('on')} style={filter==='on'?{background:'#16A34A',borderColor:'#16A34A'}:{}}>Active</Chip>
          <Chip active={filter==='off'} onClick={()=>setFilter('off')}>Off</Chip>
        </div>
      </div>
      <div style={{padding:'0 12px 20px',display:'grid',gridTemplateColumns:'repeat(3,minmax(0,1fr))',gap:10,alignItems:'start'}}>
        {squads.map(sq => {
          const sqApps = sq.cats.flatMap(c=>c.apps);
          const sqVisible = sqApps.filter(visible).length;
          if (!edit && sqVisible===0 && (ql || filter!=='all')) return null;
          return (
            <div key={sq.key} style={{background:'var(--card)',border:'1.5px solid '+sq.color+'2E',borderRadius:14,padding:8,alignSelf:'stretch'}}>
              <div style={{display:'flex',alignItems:'center',gap:5,flexWrap:'wrap',background:sq.color+'0F',border:'1.5px solid '+sq.color+'2E',borderRadius:10,padding:'6px 8px',marginBottom:8}}>
                <span style={{width:20,height:20,borderRadius:'50%',background:sq.color,display:'inline-flex',alignItems:'center',justifyContent:'center',fontSize:10,flexShrink:0}}>{sq.icon}</span>
                <span style={{fontSize:9.5,fontWeight:800,textTransform:'uppercase',letterSpacing:'.8px',color:sq.color,whiteSpace:'nowrap'}}>{sq.name}</span>
                <span style={{marginLeft:'auto',fontSize:9,fontWeight:700,color:sq.color}}>{sqApps.filter(a=>a.on).length}/{sqApps.length}</span>
                {edit && <button onClick={()=>addCat(sq.key)} style={{border:'1px dashed '+sq.color+'66',background:'transparent',color:sq.color,borderRadius:7,padding:'2px 7px',fontSize:8.5,fontWeight:700,cursor:'pointer',fontFamily:'var(--font-body)',whiteSpace:'nowrap'}}>＋ Cat</button>}
              </div>
              <div style={{display:'flex',flexDirection:'column',gap:12}}>
                {sq.cats.map(g => {
                  const apps = g.apps.filter(visible);
                  if (!edit && apps.length===0 && ql) return null;
                  return (
                    <div key={g.id}>
                      <div style={{display:'flex',alignItems:'center',gap:6,marginBottom:9,borderBottom:'2px solid '+g.color+'33',paddingBottom:5}}>
                        <span style={{fontSize:10,fontWeight:700,textTransform:'uppercase',letterSpacing:'1px',color:g.color}}>{g.name}</span>
                        <span style={{fontSize:9,color:'var(--text2)'}}>{g.apps.filter(a=>a.on).length}/{g.apps.length}</span>
                        {edit && <button onClick={()=>addApp(sq.key, g.id)} style={{marginLeft:'auto',border:'1px dashed '+g.color+'66',background:'transparent',color:g.color,borderRadius:7,padding:'2px 9px',fontSize:9,fontWeight:700,cursor:'pointer',fontFamily:'var(--font-body)'}}>＋ App</button>}
                      </div>
                      <div style={{display:'grid',gridTemplateColumns:'repeat(2,minmax(0,1fr))',gap:'10px 4px'}}>
                        {apps.map(a => (
                          <MATile key={a.id} app={a} edit={edit} groupColor={g.color}
                            onLaunch={()=>{}}
                            onToggle={()=>mutateCat(sq.key, g.id, c=>({...c, apps:c.apps.map(x=>x.id===a.id?{...x,on:!x.on}:x)}))}
                            onRename={()=>{ const n=window.prompt('Rename app', a.name); if(n&&n.trim()) mutateCat(sq.key, g.id, c=>({...c, apps:c.apps.map(x=>x.id===a.id?{...x,name:n.trim()}:x)})); }}
                            onRemove={()=>mutateCat(sq.key, g.id, c=>({...c, apps:c.apps.filter(x=>x.id!==a.id)}))} />
                        ))}
                        {apps.length===0 && <div style={{gridColumn:'1/-1',fontSize:10,color:'var(--text2)',opacity:.7,padding:'6px 0'}}>{ql?'No matches in this category.':'No apps here yet.'}</div>}
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}

Object.assign(window, { MyApps });
