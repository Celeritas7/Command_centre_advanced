// Overlap detection: [start,end) minute ranges
function findClashes(ints){
  var ids={};
  for(var a=0;a<ints.length;a++)for(var b=a+1;b<ints.length;b++){
    var s1=toMin(ints[a].time),e1=s1+ints[a].duration,s2=toMin(ints[b].time),e2=s2+ints[b].duration;
    if(s1<e2&&s2<e1){ids[ints[a].id]=true;ids[ints[b].id]=true}
  }
  return ids;
}
function toMin(t){var p=t.split(':');return parseInt(p[0])*60+parseInt(p[1]||0)}
function fmtMin(m){return String(Math.floor(m/60)).padStart(2,'0')+':'+String(m%60).padStart(2,'0')}
// Group overlapping interrupts into connected clusters
function clashGroups(ints){
  var groups=[];
  ints.forEach(function(x){
    var s=toMin(x.time),e=s+x.duration,placed=false;
    for(var g=0;g<groups.length;g++){
      if(groups[g].some(function(y){var s2=toMin(y.time),e2=s2+y.duration;return s<e2&&s2<e})){groups[g].push(x);placed=true;break}
    }
    if(!placed)groups.push([x]);
  });
  return groups.filter(function(g){return g.length>1});
}

function renderTimeline(plan,ints,clashes){

  h+='</div></div>';
  // Base row
  h+='<div class="tl-row"><div class="tl-label">BASE</div><div class="tl-track">';
  if(plan.base)h+='<div class="tl-bar '+sqCls(baseSq)+'" style="left:0%;width:100%;opacity:.25">'+(baseSq?baseSq.ico:'')+' '+esc(plan.base)+' (all day)</div>';
  ints.forEach(function(x){h+=tlBar(x,clashes)});
  h+='</div></div>';
  // One row per unique squad
  var seen={};
  ints.forEach(function(intr){
    if(seen[intr.squad])return;seen[intr.squad]=true;
    var sq=findSq(intr.squad);
    h+='<div class="tl-row"><div class="tl-label"><span>'+(sq?sq.ico:'')+'</span><span style="font-size:9px;font-weight:600;color:var(--text2)">'+esc(intr.squad)+'</span></div><div class="tl-track">';

    h+='</div></div>';
  });
  // Single now-line overlay across the whole body
  var nH=new Date().getHours(),nM=new Date().getMinutes();
  if(nH>=5&&nH<=23)h+='<div class="tl-now" style="left:calc(60px + (100% - 60px) * '+(((nH-5+nM/60)/18).toFixed(4))+')"></div>';
  h+='</div></div>';
  return h;
}

  if(wp<4.7)return '<div class="tl-tick '+cls+clash+'" style="left:'+sp.toFixed(2)+'%" title="'+tip+'"></div>'; // ~30px on a 640px track
  return '<div class="tl-bar '+cls+clash+'" style="left:'+sp.toFixed(2)+'%;width:'+wp.toFixed(2)+'%" title="'+tip+'">'+x.time+'</div>';
}
