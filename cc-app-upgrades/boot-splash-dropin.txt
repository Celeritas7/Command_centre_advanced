(function(){
  var el=document.getElementById('ccSplash'),t0=Date.now(),hold=false,pct=0;
  var MSGS=['Scanning the pitch…','Squads located','Possession mapped'],mi=0;
  var mT=setInterval(function(){ if(mi<MSGS.length-1){mi++;var m=document.getElementById('ccSplashMsg');m.textContent=MSGS[mi];m.style.animation='none';void m.offsetWidth;m.style.animation='ccRise .35s ease both';if(mi>=1)document.getElementById('ccSeg1').className='cc-seg on';if(mi>=2)document.getElementById('ccSeg2').className='cc-seg on';} },1000);
  var pT=setInterval(function(){ pct=Math.min(pct+Math.ceil(Math.random()*4)+1,100);document.getElementById('ccSplashPct').textContent=pct; },90);
  function out(){ if(!el)return;clearInterval(mT);clearInterval(pT);el.classList.add('cc-out');setTimeout(function(){el.remove();el=null;},600); }
  var auto=setTimeout(function(){ if(!hold)out(); },2850);
  window.ccSplashHold=function(){ hold=true;clearTimeout(auto); };
  window.ccSplashDone=function(){ var wait=Math.max(0,1200-(Date.now()-t0));setTimeout(out,wait); };
})();
